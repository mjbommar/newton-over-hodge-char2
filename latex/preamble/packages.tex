%% ============================================================
%% preamble/packages.tex — encodings, core packages, bibliography system
%%
%% Purpose      : loads general-purpose packages the paper needs plus the
%%                bibliography system selected in paper.yaml
%%                (citations.system). NO hyperref here (it loads last, in
%%                hyperref-last.tex, followed only by cleveref).
%% Dependencies : generated/metadata.tex (\ifPaperBib*, \ifPaperModule*,
%%                \PaperBibStyle, \PaperBibTitle, \ifPaperSortCites).
%%
%% HOUSE RULE (amsart / arXiv mathematics): this preamble stays close to
%% what a math.CO preprint actually loads. Deliberately ABSENT, and not to
%% be reintroduced without a decision record:
%%
%%   geometry  — amsart sets its own measure; a math paper's short line
%%               length is part of the look. Do not force 1in margins.
%%   titlesec  — amsart's small-caps centred headings ARE the convention.
%%   fancyhdr  — amsart already supplies AMS running heads.
%%   titling   — the AMS front matter replaces the custom title block.
%%   setspace  — mathematics is set single-spaced.
%%   eso-pic   — title-page background images are a whitepaper device.
%%   tcolorbox — coloured callouts are a whitepaper device; a theorem,
%%               a remark, or a plain paragraph carries the same content.
%% ============================================================

% --- Programming/conditional helpers (needed by every later module) ---
\usepackage{etoolbox}
\usepackage{iftex}

% --- Encoding (engine-conditional) -----------------------------------
% pdfTeX needs 8-bit font + input encoding; the Unicode engines don't
% (fontspec handles it). fonts.tex loads the actual typefaces.
\ifPDFTeX
  \usepackage[T1]{fontenc}
  \usepackage[utf8]{inputenc}
\fi

% --- Language & quoting ----------------------------------------------
\usepackage[american]{babel}
\usepackage{csquotes}   % \enquote, biblatex integration; MUST precede biblatex

% --- Math -------------------------------------------------------------
% amsart loads amsmath itself; amssymb/mathtools/amsthm are the standard
% companions. amsthm detects the AMS class and only adds what is missing.
\usepackage{amssymb}
\usepackage{mathtools}
\usepackage{amsthm}

% --- Graphics, floats -------------------------------------------------
\usepackage{graphicx}
\graphicspath{{figures/}}          % sections write \includegraphics{name} only
\usepackage{float}
\usepackage{rotating}              % sidewaystable/sidewaysfigure (wide floats)
\usepackage{pdflscape}             % landscape pages that rotate in the PDF
% xcolor is still loaded (hyperref and some table packages expect it to be
% available), but preamble/colors.tex defines a MONOCHROME token set and
% nothing in the paper prints in colour. See that file.
\usepackage{xcolor}
\usepackage[font=small,labelfont=bf,skip=6pt]{caption}
\usepackage{subcaption}

% --- Tables -----------------------------------------------------------
\usepackage{booktabs}              % the ONLY table rules sections may use
\usepackage{tabularx}
\usepackage{longtable}             % multi-page tables (with booktabs rules)
\usepackage{threeparttable}        % table notes bound to the tabular width
\usepackage{multirow}
\usepackage{makecell}

% --- Lists ------------------------------------------------------------
\usepackage{enumitem}

% --- Units (module: siunitx) -----------------------------------------
\ifPaperModuleSI
  \usepackage{siunitx}
  \sisetup{group-separator={,}, group-minimum-digits=4,
           detect-weight=true, detect-family=true}
\fi

% ======================================================================
% Bibliography system — dispatched on \ifPaperBibNatbib / \ifPaperBibBiblatex
% (paper.yaml citations.system). One \PrintBibliography macro renders
% either; sections cite with \cite/\citep/\citet (both systems support these).
%
%   natbib   — arXiv-robust: version-agnostic .bbl, trivial to ship.
%   biblatex — authoryear + biber, or numeric (needs a .bbl whose
%              format matches arXiv's TeX Live; see docs/guides/SUBMISSION.md).
% ======================================================================
\ifPaperBibNatbib
  % numeric -> numbers+sort&compress (plainnat look); authoryear -> round.
  \ifPaperBibNumeric
    \ifPaperSortCites
      \usepackage[numbers,sort&compress,square,comma]{natbib}
    \else
      \usepackage[numbers,square,comma]{natbib}
    \fi
    \newcommand{\PaperBibStyleFile}{plainnat}
  \else
    \usepackage[authoryear,round]{natbib}
    \newcommand{\PaperBibStyleFile}{plainnat}
  \fi
  % GOTCHA (amsart): amsart.cls redefines \bibliographystyle to write
  % \bibstyle to the .aux IMMEDIATELY, discarding the \AtBeginDocument
  % deferral that latex.ltx uses. A preamble-time \bibliographystyle is
  % therefore written while the .aux is still closed and is silently lost;
  % bibtex then dies with "I found no \bibstyle command" and no LaTeX
  % error is reported. Deferring it ourselves is the fix.
  \AtBeginDocument{\bibliographystyle{\PaperBibStyleFile}}
  % One renderer, both systems. natbib: a \bibliography call over bib/.
  \newcommand{\PrintBibliography}{%
    \phantomsection\addcontentsline{toc}{section}{\PaperBibTitle}%
    \renewcommand{\refname}{\PaperBibTitle}%
    \bibliography{bib/references}}
\else
  % biblatex path. natbib=true provides \citep/\citet/\citeauthor so the
  % SAME section source compiles under either bibliography system.
  \ifPaperBibNumeric
    \usepackage[backend=biber,style=numeric-comp,sorting=nyt,natbib=true,
                url=true,doi=true,maxbibnames=99]{biblatex}
  \else
    \usepackage[backend=biber,style=authoryear,sorting=nyt,natbib=true,
                maxcitenames=2,maxbibnames=99,url=true,doi=true]{biblatex}
  \fi
  \setlength{\bibitemsep}{0.4\baselineskip}
  \addbibresource{bib/references.bib}
  \newcommand{\PrintBibliography}{%
    \printbibliography[heading=bibintoc,title={\PaperBibTitle}]}
\fi

% Let biblatex break long URLs cleanly at any character rather than gap a
% justified line. We set only the break PENALTIES (robust across biblatex
% versions); the \biburl*skip muskips are left at their defaults, since
% \setlength rejects their mu units under current biblatex.
\ifPaperBibBiblatex
  \setcounter{biburlnumpenalty}{9000}
  \setcounter{biburlucpenalty}{9500}
  \setcounter{biburllcpenalty}{9500}
\fi
